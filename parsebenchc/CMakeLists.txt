cmake_minimum_required(VERSION 3.21)
project(parsebenchc LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(PARSEBENCHC_LTO "Optimize across C translation units" ON)
add_executable(parsebenchc standalone/main.c standalone/generated/scorers.c standalone/vendor/utf8proc.c)
target_include_directories(parsebenchc PRIVATE standalone)
target_compile_definitions(parsebenchc PRIVATE UTF8PROC_STATIC)
find_package(Threads REQUIRED)
target_link_libraries(parsebenchc PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(parsebenchc PRIVATE $<$<CONFIG:Release>:/O2> /fp:strict /utf-8 /bigobj)
  target_link_options(parsebenchc PRIVATE /STACK:67108864)
  find_file(PCRE2_RUNTIME NAMES libpcre2-8-0.dll
    PATHS "C:/Program Files/Git/mingw64/bin" "C:/Program Files/Tesseract-OCR")
  if(PCRE2_RUNTIME)
    add_custom_command(TARGET parsebenchc POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${PCRE2_RUNTIME}" "$<TARGET_FILE_DIR:parsebenchc>")
  endif()
else()
  target_compile_options(parsebenchc PRIVATE -O3 -ffp-contract=off)
  target_link_libraries(parsebenchc PRIVATE m)
endif()
if(PARSEBENCHC_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT HAVE_IPO OUTPUT IPO_REASON)
  if(HAVE_IPO)
    set_property(TARGET parsebenchc PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
  endif()
endif()
